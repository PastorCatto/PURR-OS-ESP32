cmake_minimum_required(VERSION 3.16)

# Windows: force ninja to pass long compiler command lines through a response
# file instead of the process command line.
#
# CreateProcess caps a command line at 32,768 bytes. The final ELF target
# (purr_os.elf.dir/project_elf_src_*.c.obj) is compiled with the PUBLIC include
# directories of EVERY component in the project, so its command line grows with
# each component added — and this project had quietly reached ~32.4KB, close
# enough that renaming one module directory from `game_mode` to `speed_demon`
# (+2 characters, on every occurrence of that path) pushed it over. The build
# then failed with:
#
#   ninja: fatal: CreateProcess: The parameter is incorrect.
#
# which names neither the limit nor the file, on a target that had built all
# day. Diagnosing it from that message alone is genuinely hard, so this is here
# to make sure nobody has to do it twice.
#
# Cheap and portable: it only changes how the argument list is delivered, and
# the variable is inert on generators that do not need it.
set(CMAKE_NINJA_FORCE_RESPONSE_FILE ON CACHE BOOL
    "Pass long command lines via response files (required on Windows)" FORCE)

# Pull in all registered modules/drivers/apps as IDF extra components.
# Generated by modulestrap — re-run 'modulestrap build all' to update.
include(${CMAKE_SOURCE_DIR}/../cattobaked/components_manifest.cmake
        OPTIONAL RESULT_VARIABLE _manifest_loaded)
if(NOT _manifest_loaded)
    message(WARNING
        "cattobaked/components_manifest.cmake not found — "
        "run 'python3 modulestrap/modulestrap.py build all' from repo root")
endif()

set(PURR_DEVICE $ENV{PURR_DEVICE})

# EXCLUDE_COMPONENTS is gone, and deliberately.
#
# It existed to compensate for modulestrap emitting EVERY discovered component
# into EVERY device's build: components that could not compile for a given
# target (m5tab5_bsp needs MIPI-DSI hardware; nougat is v9-only source) had to
# be denied by hand, per device, in this file.
#
# That made three places decide what a device contains — device.pcat via the
# generated glue, per-component Kconfig gating, and this denylist — with none
# authoritative. They could and did disagree: tab5's device.pcat selected
# Cupcake while this list removed the component, so the glue referenced a module
# that was never compiled ("undefined reference to purr_module_cupcake") and the
# device shipped stale binaries for a week.
#
# modulestrap now generates components_manifest.cmake per device, from
# device.pcat plus the transitive REQUIRES of what it names. A component that
# cannot build for a target is simply never added, so there is nothing to
# exclude, and a contradiction of that shape can no longer be expressed.

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

# Force LVGL's allocator onto PSRAM instead of the board's scarce internal
# SRAM — see source/config/lv_conf.h for the full rationale. LV_CONF_PATH is
# honored by lv_conf_internal.h ahead of any Kconfig fallback (as long as
# CONFIG_LV_CONF_SKIP=n — see sdkconfig_tdeck_plus.overrides, it defaults to
# y and silently skips this whole mechanism otherwise), and setting it as a
# global build property (rather than a per-component include path) applies
# it to every component, including the vendored lvgl__lvgl itself.
#
# Quoting differs by LVGL major version — confirmed by reading each
# version's own lv_conf_internal.h, not assumed to be the same mechanism:
#   v8: `#include __LV_TO_STR(LV_CONF_PATH)` — __LV_TO_STR stringizes (#x)
#       its argument itself. A pre-quoted value here double-stringizes into
#       a malformed nested-quote path ("\"C:/...\"") that GCC rejects with
#       "fatal error: ...: Invalid argument" — confirmed live. Must stay
#       the bare, unquoted path.
#   v9: `#include LV_CONF_PATH` — no wrapping macro at all (the header's
#       own comment says so: "Make sure to define custom CONF_PATH as a
#       string"). An unquoted path here is not a valid #include operand at
#       all — "#include expects FILENAME or <FILENAME>" — confirmed live.
#       Must be pre-quoted.
#
# LVGL v8 and v9 have different lv_conf.h schemas, so this used to be
# PURR_DEVICE-conditional: tab5 got lv_conf_v9.h with v9-style pre-quoting,
# everything else got lv_conf.h bare.
#
# CORRECTION (2026-07-28): that split was for a v9 resolution that does not
# exist. CoreOS/main/idf_component.yml pins `lvgl/lvgl: ">=8.3.0,<9.0.0"` for
# every target with no rules clause, managed_components is shared across
# devices, and the resolved component is **8.4.0** — verified in
# dependencies.lock and lvgl.h's own LVGL_VERSION_MAJOR.
#
# So tab5 was being handed a v9-schema config, with v9 quoting, against a v8
# library. Every device now takes the v8 branch, which is what every device
# actually links. Restore the split only if a device genuinely resolves v9 —
# and then fix the manifest first, because today nothing can.
get_filename_component(PURR_LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/../source/config/lv_conf.h" ABSOLUTE)
idf_build_set_property(COMPILE_DEFINITIONS "LV_CONF_PATH=${PURR_LV_CONF_PATH}" APPEND)

project(purr_os)
